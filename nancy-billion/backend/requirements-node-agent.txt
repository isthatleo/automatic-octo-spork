# node_agent_stub.py's own dependencies -- NOT part of requirements.txt on
# purpose. That file installs into the Linux backend container; this script
# runs natively on the Windows machine it's paired with (see
# scripts/start-node-agent.ps1), a completely separate Python environment
# with its own footprint. Install with:
#

#
# fastapi/uvicorn/pydantic are already needed for the stub's own HTTP server
# but are pinned loosely here since this environment doesn't share the
# backend container's exact-version lock file.

PyAutoGUI==0.9.54
